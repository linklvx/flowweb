import { Injectable } from '@nestjs/common';

@Injectable()
export class SubscriptionBannerService {
  constructor(
    private readonly prisma: any,
    private readonly redis: any,
  ) {}

  async getPublicBanner() {
    // 先查缓存
    const cached = await this.redis.get('subscription:banner:public');
    if (cached) return JSON.parse(cached);

    const banner = await this.prisma.subscriptionBanner.findFirst();
    if (!banner || !banner.isActive) {
      // 空值缓存防穿透
      await this.redis.set('subscription:banner:public', 'null', 'EX', 60);
      return null;
    }

    const result = {
      title: banner.title,
      subtitle: banner.subtitle,
      backgroundImageKey: banner.backgroundImageKey,
      backgroundImageUrl: banner.backgroundImageUrl,
      countdownEndAt: banner.countdownEndAt?.toISOString() ?? null,
    };

    await this.redis.set(
      'subscription:banner:public',
      JSON.stringify(result),
      'EX',
      60,
    );
    return result;
  }
}

- [ ] **Step 7: 实现自动延期逻辑**

更新 `subscription-banner.service.ts` 的 `getPublicBanner` 方法：

```ts
import { Injectable } from '@nestjs/common';
import { randomUUID } from 'crypto';

@Injectable()
export class SubscriptionBannerService {
  constructor(
    private readonly prisma: any,
    private readonly redis: any,
  ) {}

  async getPublicBanner() {
    const cached = await this.redis.get('subscription:banner:public');
    if (cached) {
      if (cached === 'null') return null;
      return JSON.parse(cached);
    }

    const banner = await this.prisma.subscriptionBanner.findFirst();
    if (!banner || !banner.isActive) {
      await this.redis.set('subscription:banner:public', 'null', 'EX', 60);
      return null;
    }

    // Auto-extend check
    if (
      banner.autoExtend &&
      banner.countdownEndAt &&
      banner.countdownEndAt < new Date()
    ) {
      const lockKey = 'subscription:banner:auto-extend:lock';
      const lockValue = randomUUID();
      const locked = await this.redis.set(lockKey, lockValue, 'NX', 'PX', 1000);

      if (locked === 'OK') {
        // 基于原截止时间 +3天
        const newEndAt = new Date(
          banner.countdownEndAt.getTime() + 3 * 24 * 60 * 60 * 1000,
        );
        await this.prisma.subscriptionBanner.update({
          where: { id: banner.id },
          data: { countdownEndAt: newEndAt },
        });
        // Lua脚本原子释放锁（校验value匹配）
        await this.redis.eval(
          "if redis.call('get', KEYS[1]) == ARGV[1] then return redis.call('del', KEYS[1]) else return 0 end",
          1,
          lockKey,
          lockValue,
        );
        banner.countdownEndAt = newEndAt;
        // 延期后删除缓存
        await this.redis.del('subscription:banner:public');
      }
    }

    const result = {
      title: banner.title,
      subtitle: banner.subtitle,
      backgroundImageKey: banner.backgroundImageKey,
      backgroundImageUrl: banner.backgroundImageUrl,
      countdownEndAt: banner.countdownEndAt?.toISOString() ?? null,
    };

    await this.redis.set(
      'subscription:banner:public',
      JSON.stringify(result),
      'EX',
      60,
    );
    return result;
  }
}
```

  const banner = await this.prisma.subscriptionBanner.findFirst();

async updateBanner(dto: any) {
  const existing = await this.prisma.subscriptionBanner.findFirst();

  // 联动校验
  if (dto.autoExtend === true) {
    const willHave = dto.countdownEndAt ?? existing?.countdownEndAt;
    if (!willHave) {
      throw new Error('countdownEndAt is required when autoExtend is enabled');
    }
  }

  const updated = await this.prisma.subscriptionBanner.upsert({
    where: { id: 'subscription-banner-singleton' },
    create: {
      id: 'subscription-banner-singleton',
      title: dto.title ?? '',
      subtitle: dto.subtitle ?? '',
      ...dto,
    },
    update: dto,
  });

  // 删除缓存
  await this.redis.del('subscription:banner:public');

  return updated;
}

import { Controller, Get, Patch, Body, Post, UploadedFile, UseInterceptors } from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { SubscriptionBannerService } from './subscription-banner.service';

@Controller()
export class SubscriptionBannerController {
  constructor(private readonly bannerService: SubscriptionBannerService) {}

  // Public: GET /api/subscription/banner
  @Get('api/subscription/banner')
  async getPublicBanner() {
    return this.bannerService.getPublicBanner();
  }

  // Admin: GET /api/admin/subscription/banner
  @Get('api/admin/subscription/banner')
  async getAdminBanner() {
    return this.bannerService.getAdminBanner();
  }

  // Admin: PATCH /api/admin/subscription/banner
  @Patch('api/admin/subscription/banner')
  async updateBanner(@Body() body: any) {
    // XSS过滤：trim strings
    if (body.title) body.title = body.title.trim().slice(0, 64);
    if (body.subtitle) body.subtitle = body.subtitle.trim().slice(0, 200);
    if (body.backgroundImageUrl) {
      const url = body.backgroundImageUrl.trim();
      // 协议白名单 + 防SSRF + XSS
      if (!/^https?:\/\//.test(url)) {
        throw new Error('Invalid backgroundImageUrl: only http/https allowed');
      }
      body.backgroundImageUrl = url.slice(0, 500);
    }
    return this.bannerService.updateBanner(body);
  }

  // Admin: POST /api/admin/subscription/banner/upload
  @Post('api/admin/subscription/banner/upload')
  @UseInterceptors(FileInterceptor('file', {
    limits: { fileSize: 2 * 1024 * 1024 }, // 2MB
    fileFilter: (_req, file, cb) => {
      const allowed = ['image/jpeg', 'image/png', 'image/webp'];
      if (!allowed.includes(file.mimetype)) {
        return cb(new Error('Invalid file type'), false);
      }
      cb(null, true);
    },
  }))

function BannerCountdown({ endAt }: { endAt: string }) {
  const calc = () => {
    const diff = new Date(endAt).getTime() - Date.now();
    if (diff <= 0) return { days: '00', hours: '00', mins: '00', secs: '00', expired: true };
    return {
      days: String(Math.floor(diff / 86400000)).padStart(2, '0'),
      hours: String(Math.floor((diff % 86400000) / 3600000)).padStart(2, '0'),
      mins: String(Math.floor((diff % 3600000) / 60000)).padStart(2, '0'),
      secs: String(Math.floor((diff % 60000) / 1000)).padStart(2, '0'),
      expired: false,
    };
  };

  const [time, setTime] = useState(calc);

  useEffect(() => {
    const timer = setInterval(() => setTime(calc()), 1000);

    const onVisible = () => {
      const t = calc();
      setTime(t);
      // 如果已过期，触发父组件刷新逻辑（可选：通过回调）
    };
    document.addEventListener('visibilitychange', () => {
      if (!document.hidden) onVisible();
    });

    return () => {
      clearInterval(timer);
      document.removeEventListener('visibilitychange', onVisible);
    };
  }, [endAt]);

  return (
    <div className="flex gap-3 items-center">
      {[
        { value: time.days, unit: '天' },
        { value: time.hours, unit: '时' },
        { value: time.mins, unit: '分' },
        { value: time.secs, unit: '秒' },
      ].map(({ value, unit }) => (
        <div key={unit} className="flex flex-col items-center">
          <span className="font-mono text-2xl font-bold text-white bg-[#ffffff15] rounded-lg px-3 py-1 min-w-[48px] text-center">
            {value}
          </span>
          <span className="text-xs text-[#888] mt-1">{unit}</span>
        </div>
      ))}
    </div>
  );
}
